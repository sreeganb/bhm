\section{Introduction}
\label{sec:introduction}

In this document, we describe a Bayesian hierarchical modeling framework to assemble protein complexes 
using experimental data, such as crosslinking mass spectrometry and cryo-electron microscopy (cryo-EM). 
This approach exploits the hierarchical nature of protein complex assembly and integrates multiple data sources 
to infer the spatial structure of the complex. We elaborate on the mathematical formulation with additional equations 
and illustrate the method with a toy model problem based on distance restraints.

\section{Hierarchical Assembly Structure}
\label{sec:hierarchical_assembly}

Protein complex assembly can be represented as a tree:
\begin{itemize}
    \item \textbf{Leaves}: Individual proteins (e.g., $P_1, P_2, \ldots, P_n$).
    \item \textbf{Internal Nodes}: Subcomplexes formed by combining their children (e.g., $C_1$ formed by $P_1$ and $P_2$).
    \item \textbf{Root}: The fully assembled complex.
\end{itemize}

For example, a complex $C$ might consist of a subcomplex $C_1$ (formed by $P_1$ and $P_2$) and protein $P_3$, depicted as:
\[
\begin{array}{c}
C \\
/ \quad \backslash \\
C_1 \quad P_3 \\
/ \backslash \\
P_1 \quad P_2
\end{array}
\]

\section{Parameterization with Transformations}
\label{sec:parameterization}

The spatial configuration is parameterized using rigid transformations (rotations and translations):
\begin{itemize}
    \item For each edge from a parent to a child node, define a relative transformation $T_{\text{child}|\text{parent}}$.
    \item The absolute transformation of a node is the composition of relative transformations from the root to that node.
\end{itemize}

For the example above:
\begin{itemize}
    \item $T_{C_1|C}$: Positions $C_1$ relative to $C$.
    \item $T_{P_3|C}$: Positions $P_3$ relative to $C$.
    \item $T_{P_1|C_1}$ and $T_{P_2|C_1}$: Position $P_1$ and $P_2$ within $C_1$.
    \item Absolute transformation of $P_2$: $T_{P_2} = T_{C_1|C} \cdot T_{P_2|C_1}$.
\end{itemize}

A transformation $T$ can be represented as a rotation matrix $R \in SO(3)$ and a translation vector $\mathbf{t} 
\in \mathbb{R}^3$, applied to a point $\mathbf{x}$ as:
\[
\mathbf{x}' = R \mathbf{x} + \mathbf{t}
\]

\section{Prior Distributions}
\label{sec:priors}

Priors encode prior knowledge about transformations:
\begin{itemize}
    \item \textbf{Rotations}: Use a von Mises-Fisher distribution on $SO(3)$:
    \[
    p(R) \propto \exp(\kappa \cdot \text{tr}(R_0^T R))
    \]
    where $R_0$ is the mean orientation, $\kappa$ is the concentration parameter, and $\text{tr}$ is the trace.
    \item \textbf{Translations}: Use a multivariate normal distribution:
    \[
    p(\mathbf{t}) = \frac{1}{(2\pi)^{3/2} |\Sigma|^{1/2}} \exp\left(-\frac{1}{2} (\mathbf{t} - \mathbf{t}_0)^T \Sigma^{-1} (\mathbf{t} - \mathbf{t}_0)\right)
    \]
    where $\mathbf{t}_0$ is the mean translation and $\Sigma$ is the covariance matrix.
\end{itemize}

For tight subcomplexes, $\kappa$ is large and $\Sigma$ has small variances; for looser associations, they are broader.

\section{Likelihood Based on Experimental Data}
\label{sec:likelihood}

The likelihood quantifies how well the model fits the data:
\begin{itemize}
    \item \textbf{Crosslinking Mass Spectrometry}: Provides distance restraints. For a crosslink between residues $i$ and $j$ with positions $\mathbf{x}_i$ and $\mathbf{x}_j$, the distance is:
    \[
    d_{ij} = \|\mathbf{x}_i - \mathbf{x}_j\|_2
    \]
    The likelihood is:
    \[
    P(\text{crosslink}_{ij} | T) = \frac{1}{\sqrt{2\pi} \sigma} \exp\left(-\frac{(d_{ij} - d_0)^2}{2\sigma^2}\right)
    \]
    where $d_0$ is the expected distance and $\sigma$ is the uncertainty.
    \item \textbf{Cryo-EM}: Provides a density map $\rho_{\text{obs}}$. The predicted density $\rho_{\text{pred}}(T)$ depends on the transformations. The likelihood can use cross-correlation (CC):
    \[
    \text{CC} = \frac{\int \rho_{\text{obs}}(\mathbf{r}) \rho_{\text{pred}}(\mathbf{r}) d\mathbf{r}}{\sqrt{\int \rho_{\text{obs}}^2(\mathbf{r}) d\mathbf{r} \int \rho_{\text{pred}}^2(\mathbf{r}) d\mathbf{r}}}
    \]
    \[
    P(\text{density} | T) \propto \exp\left(-\frac{(1 - \text{CC})^2}{2\sigma_{\text{EM}}^2}\right)
    \]
\end{itemize}

Total likelihood:
\[
P(\text{data} | T) = \prod_{\text{crosslinks}} P(\text{crosslink}_{ij} | T) \cdot P(\text{density} | T)
\]

\section{Bayesian Inference}
\label{sec:bayesian_inference}

The posterior distribution combines the likelihood and prior:
\[
P(T | \text{data}) = \frac{P(\text{data} | T) P(T)}{P(\text{data})}
\]
Since $P(\text{data})$ is a normalizing constant, we sample from:
\[
P(T | \text{data}) \propto P(\text{data} | T) P(T)
\]
using methods like Markov Chain Monte Carlo (MCMC).

The log-posterior, useful for optimization, is:
\[
\log P(T | \text{data}) = \log P(\text{data} | T) + \log P(T) + \text{constant}
\]

\section{Toy Model Problem}
\label{sec:toy_model}

Consider a complex with three proteins: $P_1$, $P_2$, and $P_3$, where $P_1$ and $P_2$ form subcomplex $C_1$, and $C_1$ joins $P_3$ to form $C$:
\[
\begin{array}{c}
C \\
/ \quad \backslash \\
C_1 \quad P_3 \\
/ \backslash \\
P_1 \quad P_2
\end{array}
\]

\subsection{Experimental Data}
\label{subsec:experimental_data}
Distance restraints from crosslinking:
\begin{itemize}
    \item $P_1$ (residue at $\mathbf{x}_{P_1}$) to $P_2$ (residue at $\mathbf{x}_{P_2}$): $d_1 = 10$ \AA, $\sigma_1 = 1$ \AA.
    \item $P_2$ (residue at $\mathbf{x}_{P_2}$) to $P_3$ (residue at $\mathbf{x}_{P_3}$): $d_2 = 15$ \AA, $\sigma_2 = 1.5$ \AA.
\end{itemize}

\subsection{Parameterization}
\label{subsec:toy_parameterization}
Define transformations:
\begin{itemize}
    \item $T_{C_1|C} = (R_{C_1|C}, \mathbf{t}_{C_1|C})$: $C_1$ relative to $C$.
    \item $T_{P_3|C} = (R_{P_3|C}, \mathbf{t}_{P_3|C})$: $P_3$ relative to $C$.
    \item $T_{P_1|C_1} = (R_{P_1|C_1}, \mathbf{t}_{P_1|C_1})$: $P_1$ relative to $C_1$.
    \item $T_{P_2|C_1} = (R_{P_2|C_1}, \mathbf{t}_{P_2|C_1})$: $P_2$ relative to $C_1$.
\end{itemize}

Absolute positions:
\begin{itemize}
    \item $\mathbf{x}_{P_1} = T_{C_1|C} T_{P_1|C_1} \mathbf{x}_{P_1}^0$.
    \item $\mathbf{x}_{P_2} = T_{C_1|C} T_{P_2|C_1} \mathbf{x}_{P_2}^0$.
    \item $\mathbf{x}_{P_3} = T_{P_3|C} \mathbf{x}_{P_3}^0$.
\end{itemize}
where $\mathbf{x}^0$ are coordinates in the protein’s reference frame.

\subsection{Priors}
\label{subsec:toy_priors}
\begin{itemize}
    \item $R_{P_1|C_1}$, $R_{P_2|C_1}$: von Mises-Fisher with $\kappa = 10$, centered at identity.
    \item $\mathbf{t}_{P_1|C_1}$, $\mathbf{t}_{P_2|C_1}$: $\mathcal{N}(\mathbf{0}, \sigma_t^2 I)$, $\sigma_t = 2$ \AA.
    \item $T_{C_1|C}$, $T_{P_3|C}$: Broader priors (e.g., $\kappa = 1$, $\sigma_t = 5$ \AA).
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}

This framework provides a robust method for protein complex assembly, integrating hierarchical modeling with experimental data. The toy model demonstrates its application to simple systems, extensible to larger complexes with additional data.
