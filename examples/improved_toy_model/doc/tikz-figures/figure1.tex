  ]

  % Define node styles
  \tikzstyle{hyper} = [latent, minimum size=20pt, fill=gray!20]
  \tikzstyle{latent_main} = [latent, minimum size=22pt]
  \tikzstyle{latent_sub} = [latent, minimum size=20pt]
  \tikzstyle{det_node} = [det, minimum size=20pt]
  \tikzstyle{obs_node} = [obs, minimum size=22pt]

  % Top-level Hyperparameters
  \node[hyper] (phi) {$\Phi$};
  \node[hyper, right=of phi] (psi) {$\Psi$};

  % Level 1: Global System Variables
  \node[latent_main, below=of phi] (C) {$C$};
  \node[latent_main, below=of psi] (X_complex) {$X_{\text{complex}}$};

  % Level 2: Sub-system and Nuisance Parameters
  % Plate for N Subunits
  \node[latent_sub, below left=of X_complex, xshift=-1cm] (T_i) {$T_i$};
  \node (plate_subunits) {};
  
  % Deterministic Subcomplex node
  \node[det_node, below right=of X_complex, xshift=1cm] (X_sub) {$X_{\text{sub}}$};

  % Nuisance parameters for data
  \node[latent_sub, below=of C] (theta_em) {$\Theta_{\text{EM}}$};
  \node[latent_sub, left=of X_sub] (theta_nmr) {$\Theta_{\text{NMR}}$};
  \node[latent_sub, right=of X_sub] (theta_xl) {$\Theta_{\text{XL}}$};

  % Level 3: Data
  \node[obs_node, below=of theta_em] (D_em) {$D_{\text{EM}}$};
  \node[obs_node, below=of theta_nmr] (D_nmr) {$D_{\text{NMR}}$};
  \node[obs_node, below=of theta_xl] (D_xl) {$D_{\text{XL}}$};

  % Plate for NMR restraints
  \node (plate_nmr) {};
  % Plate for XL-MS restraints
  \node (plate_xl) {};

  % Edges (Dependencies)
  \edge {phi} {C}; % Prior on Stoichiometry
  \edge {phi} {theta_em, theta_nmr, theta_xl}; % Priors on nuisance params

  \edge {psi} {X_complex}; % Prior on structure (e.g., force field)
  
  \edge {X_complex} {T_i}; % Global transformations determine subunit transforms
  \edge {X_complex} {X_sub}; % Subcomplex structure is function of global
  
  \edge {C, X_complex, theta_em} {D_em}; % EM data depends on stoich, structure, nuisance
  
  \edge {X_sub, theta_nmr} {D_nmr}; % NMR data depends on subcomplex structure
  \edge {X_complex, theta_xl} {D_xl}; % XL-MS data depends on global structure

\end{tikzpicture}
